String [] getCadenas(String [] cadenas, Integer i){
	

	cadenas[i] = getNewCadena(i);
	
	return cadenas;
}

String getNewCadena(Integer i){

	return "Cadenita_"+(i+1);

}

String Imprimir(String [] arreglo, String tipo){
	Switch ( tipo.toLowerCase() ){

		case "tolowercase":
			for(Integer i = 0; i < arreglo.length; i++){
				print(arreglo[i].tolowercase());
			}
			break;
		
		case "touppercase":
			for(Integer i = 0; i < arreglo.length; i++){
				print(arreglo[i].toUppercase());
			}
			break;
		
		case "charat":
		
			for(Integer i = 0; i < arreglo.length; i++){
				print(arreglo[i]);
				for(Integer j = 0; j < arreglo[i].length(); j++){
					print(arreglo[i].charArt(j));
				}
			}
			break;
		
		case "length":
		
			for(Integer i = 0; i < arreglo.length; i++){
				print("Length cadena: "+arreglo[i]+" es: "+arreglo[i].length());
			}
			break;
		
	}

}
